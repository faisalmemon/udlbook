\section{Equivariance and invariance}

If $\mathbf{P}$ is a permutation matrix, then the following holds:
\begin{equation*}
    \mathbf{H}_{k+1}\mathbf{P} = \mathbf{F}[\mathbf{H}_{k}\mathbf{P}, \mathbf{P}^{T}\mathbf{A}\mathbf{\mathbf{P}}, \boldsymbol{\phi}_{k}].
\end{equation*}

\section{Problem 13.7}
Consider the simple GNN layer:
\begin{equation*}
    \mathbf{H}_{k+1} = GraphLayer[\mathbf{H}_{k}, \mathbf{A}]
        = \mathbf{a}[\boldsymbol{\beta}_{k}\mathbf{1}^{T} + \boldsymbol{\Omega}_{k}\begin{bmatrix}
            \mathbf{H}_{k}\\
            \mathbf{H}_{k}\mathbf{A}
            \end{bmatrix}]
\end{equation*}

where \(H\) is a \(D \times N\) matrix containing the \(N\) node embeddings in its columns,
 \(A\) is the \(N \times N\) adjacency matrix, \(\beta\) is the bias vector, and \(\Omega\) is the weight matrix.

Show that 
\begin{equation*}
    GraphLayer[\mathbf{H}_{k}, \mathbf{A}]\mathbf{P} = GraphLayer[\mathbf{H}_{k}\mathbf{P}, \mathbf{P}^{T}\mathbf{A}\mathbf{\mathbf{P}}]
\end{equation*}
